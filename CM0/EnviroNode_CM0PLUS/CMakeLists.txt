# Set the minimum version of CMake that can be used
cmake_minimum_required(VERSION 3.20)

# Enable compile command to ease indexing with e.g. clangd
set(CMAKE_EXPORT_COMPILE_COMMANDS TRUE)

# Project definition
project(EnviroNode_CM0PLUS)

# Enable CMake support for ASM and C/CPP languages
enable_language(C CXX ASM)

set(BUILD_UNIT_0_NAME EnviroNode_CM0PLUS)

# Create an executable object type
add_executable(${BUILD_UNIT_0_NAME})

# STM32CubeIDE for Visual Studio Code managed CMake file
include(cmake/vscode_generated.cmake)

# Compiler options
target_compile_options(${BUILD_UNIT_0_NAME} PRIVATE
  # User defined compiler options
)

# Linker options
target_link_options(${BUILD_UNIT_0_NAME} PRIVATE 
  # User defined linker options
)

# ----------------------------------------------------------------------------
# KoreroNet: LoRaWAN node sources (added by hand â€” not generated by CubeMX).
# The base skeleton sources (Core/Src/main.c, subghz.c, it.c, hal_msp,
# syscalls, sysmem, HAL drivers, startup, system) come from
# cmake/vscode_generated.cmake; the files below are layered on top.
# ----------------------------------------------------------------------------
file(GLOB KORERO_LORAWAN_SRC
  "${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/LoRaWAN/Crypto/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/LoRaWAN/LmHandler/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/LoRaWAN/LmHandler/Packages/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/LoRaWAN/Mac/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/LoRaWAN/Mac/Region/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/LoRaWAN/Utilities/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Middlewares/Third_Party/SubGHz_Phy/stm32_radio_driver/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Utilities/lpm/tiny_lpm/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Utilities/misc/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Utilities/sequencer/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Utilities/timer/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Utilities/trace/adv_trace/*.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/LoRaWAN/App/*.c"
)
# Drop the *_template.c stubs â€” the real implementations live in Core/Src.
list(FILTER KORERO_LORAWAN_SRC EXCLUDE REGEX "_template\\.c$")
# Drop FUOTA / multicast / clock-sync packages (not used by this end node; they
# pull app-level headers like frag_decoder_if.h that we don't ship). Matches the
# stock LoRaWAN_End_Node, which builds only Compliance + PackagesRegistration.
list(FILTER KORERO_LORAWAN_SRC EXCLUDE REGEX "(FragDecoder|LmhpFragmentation|LmhpFirmwareManagement|LmhpRemoteMcastSetup|LmhpClockSync)\\.c$")

# Add sources to executable/library
target_sources(${BUILD_UNIT_0_NAME} PRIVATE
  ${KORERO_LORAWAN_SRC}
  "${CMAKE_CURRENT_SOURCE_DIR}/LoRaWAN/Target/radio_board_if.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Drivers/BSP/STM32WLxx_Nucleo/stm32wlxx_nucleo_radio.c"
  # Core/platform glue (the skeleton set is already in vscode_generated.cmake)
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/sys_app.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/sys_debug.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/usart.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/usart_if.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/dma.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/gpio.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/adc_if.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/timer_if.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/stm32_lpm_if.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/sys_sensors.c"
  "${CMAKE_CURRENT_SOURCE_DIR}/Core/Src/flash_if.c"
)

# Add include directories
target_include_directories(${BUILD_UNIT_0_NAME} PRIVATE
  Core/Inc
  LoRaWAN/App
  LoRaWAN/Target
  Middlewares/Third_Party/LoRaWAN/Crypto
  Middlewares/Third_Party/LoRaWAN/LmHandler
  Middlewares/Third_Party/LoRaWAN/LmHandler/Packages
  Middlewares/Third_Party/LoRaWAN/Mac
  Middlewares/Third_Party/LoRaWAN/Mac/Region
  Middlewares/Third_Party/LoRaWAN/Utilities
  Middlewares/Third_Party/SubGHz_Phy
  Middlewares/Third_Party/SubGHz_Phy/stm32_radio_driver
  Utilities/lpm/tiny_lpm
  Utilities/misc
  Utilities/sequencer
  Utilities/timer
  Utilities/trace/adv_trace
)
